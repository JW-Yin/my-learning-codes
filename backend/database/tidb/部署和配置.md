
# 容器化部署

```yml
# pd:1
# tikv:1
# tidb:1
# tiflash:1
# grafana:1

version: '3.8'

networks:
  tidb-net:
    driver: bridge

services:
  pd:
    image: pingcap/pd:v8.5.5
    container_name: tidb-pd
    networks:
      - tidb-net
    ports:
      - "2379:2379"
    volumes:
      - pd-data:/data/pd
    command:
      - --name=pd1
      - --data-dir=/data/pd
      - --client-urls=http://0.0.0.0:2379
      - --advertise-client-urls=http://pd:2379
      - --peer-urls=http://0.0.0.0:2380
      - --advertise-peer-urls=http://pd:2380
      - --initial-cluster=pd1=http://pd:2380

  tikv:
    image: pingcap/tikv:v8.5.5
    container_name: tidb-tikv
    networks:
      - tidb-net
    volumes:
      - tikv-data:/data/tikv
    depends_on:
      - pd
    command:
      - --pd-endpoints=pd:2379
      - --addr=0.0.0.0:20160
      - --advertise-addr=tikv:20160
      - --data-dir=/data/tikv

  tiflash:
    image: pingcap/tiflash:v8.5.5
    container_name: tidb-tiflash
    networks:
      - tidb-net
    volumes:
      - tiflash-data:/data/tiflash
    depends_on:
      - pd
      - tikv
    command:
      - --server --addr=0.0.0.0:3930
      - --advertise-addr=tiflash:3930
      - --data-dir=/data/tiflash
      - --pd-endpoints=pd:2379

  tidb:
    image: pingcap/tidb:v8.5.5
    container_name: tidb-server
    networks:
      - tidb-net
    ports:
      - "4000:4000"
      - "10080:10080"
    depends_on:
      - pd
      - tikv
    command:
      - --store=tikv
      - --path=pd:2379

  grafana:
    image: grafana/grafana:latest
    container_name: tidb-grafana
    networks:
      - tidb-net
    ports:
      - "3000:3000"
    environment:
      - GF_SECURITY_ADMIN_USER=admin
      - GF_SECURITY_ADMIN_PASSWORD=admin

volumes:
  pd-data:
  tikv-data:
  tiflash-data:

```
